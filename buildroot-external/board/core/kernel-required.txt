# Kernel options every edition relies on that no fragment sets: x86_64_defconfig
# happens to provide them. A kernel bump that drops one fails the build here
# instead of shipping a machine that quietly can't do something (2026-10-17:
# PCIe hotplug had never been on, and a hybrid laptop's dGPU never powered off).
# Fragment lines are checked on their own; only list what they don't set.
# Checked by scripts/check-kernel-fragments.py --built, run from the core
# post-build. `CONFIG_FOO` = y or m, `CONFIG_FOO=y` = built in.

# --- Boot: UEFI, the ESP, the live stick, no initramfs ---------------------------
CONFIG_EFI=y                # every image boots UEFI
CONFIG_EFI_STUB=y           # GRUB hands bzImage over through the EFI stub
CONFIG_EFI_ESRT=y           # fwupd's UEFI capsule updates read the ESRT
CONFIG_VFAT_FS=y            # the ESP (installer, updater, fwupd)
CONFIG_NLS_CODEPAGE_437=y   # vfat's default codepage: without it the ESP won't mount
CONFIG_NLS_ISO8859_1=y      # vfat's default iocharset
CONFIG_USB_XHCI_HCD=y       # the live stick and keyboards on USB 3 controllers
CONFIG_USB_EHCI_HCD=y       # ...and on USB 2 ones
CONFIG_USB_STORAGE=y        # built in: the live stick's root is on it, before any module loads
CONFIG_BLK_DEV_SD=y         # USB and SATA disks show up as sd*
CONFIG_SCSI=y
CONFIG_BLK_DEV_LOOP=y       # image tooling, the installer's image writes

# --- systemd and its sandboxing -------------------------------------------------
CONFIG_CGROUPS=y
CONFIG_CGROUP_PIDS=y        # TasksMax=
CONFIG_CGROUP_DEVICE=y
CONFIG_CGROUP_SCHED=y       # CPUWeight=
CONFIG_FAIR_GROUP_SCHED=y
CONFIG_CPUSETS=y            # AllowedCPUs=
CONFIG_NAMESPACES=y         # PrivateTmp=, PrivateNetwork=, ...; Flatpak's bubblewrap
CONFIG_USER_NS=y            # unprivileged sandboxes (bubblewrap, rootless podman)
CONFIG_PID_NS=y
CONFIG_NET_NS=y
CONFIG_SECCOMP=y            # SystemCallFilter=, Flatpak, browsers' sandboxes
CONFIG_SECCOMP_FILTER=y
CONFIG_FHANDLE=y            # systemd hard requirement (name_to_handle_at)
CONFIG_INOTIFY_USER=y       # systemd hard requirement
CONFIG_SIGNALFD=y           # systemd hard requirement
CONFIG_TIMERFD=y            # systemd hard requirement
CONFIG_EPOLL=y              # systemd hard requirement
CONFIG_AUTOFS_FS=y          # systemd automount units
CONFIG_TMPFS_POSIX_ACL=y    # logind's per-seat device ACLs, journald
CONFIG_PROC_FS=y
CONFIG_SYSFS=y
CONFIG_SHMEM=y
CONFIG_SYSVIPC=y
CONFIG_POSIX_MQUEUE=y

# --- Networking -------------------------------------------------------------------
CONFIG_UNIX=y
CONFIG_INET=y
CONFIG_IPV6=y
CONFIG_PACKET=y             # DHCP clients send raw packets

# --- Hardware basics ---------------------------------------------------------------
CONFIG_SMP=y
CONFIG_PCI_MSI=y
CONFIG_PCIEPORTBUS=y        # what PCIe hotplug, AER, DPC and port PM hang off
CONFIG_HOTPLUG_PCI=y
CONFIG_IOMMU_SUPPORT=y      # DMA protection (Thunderbolt/USB4), VFIO passthrough
CONFIG_AMD_IOMMU=y
CONFIG_INTEL_IOMMU=y
CONFIG_ACPI=y
CONFIG_ACPI_BUTTON=y        # power button and lid switch
CONFIG_ACPI_BATTERY=y       # laptop battery
CONFIG_ACPI_AC=y            # on mains / on battery
CONFIG_DMI_SYSFS=y          # fwupd, hardware identification
CONFIG_RTC_CLASS=y
CONFIG_RTC_DRV_CMOS=y       # the clock across reboots
CONFIG_CPU_FREQ=y
CONFIG_X86_AMD_PSTATE=y     # AMD frequency scaling
CONFIG_X86_INTEL_PSTATE=y   # Intel frequency scaling
CONFIG_CPU_IDLE=y
CONFIG_MICROCODE=y          # the amd-ucode/intel-ucode we ship are loaded through it
CONFIG_SUSPEND=y            # suspend to RAM / s2idle
CONFIG_MAGIC_SYSRQ=y
CONFIG_INPUT_EVDEV=y        # every input device userspace sees
CONFIG_INPUT_KEYBOARD=y
CONFIG_KEYBOARD_ATKBD=y     # laptop built-in keyboards (i8042)

CONFIG_MOUSE_PS2=y          # PS/2 touchpads
