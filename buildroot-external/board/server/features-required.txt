# Feature claims for the Server image (and every edition stacked on it).
#
# The Server twin of board/desktop/features-required.txt; same format, same
# checker (scripts/check-daemon-features.py), run from the SERVER post-build, so
# the Desktop editions are held to it too. These are things nothing in the image
# links or starts, so a layer edit that drops one is silent until a user needs it.
#
# Format: `kind  argument...`  then an optional `  # why`
#   contains <path> <needle> | exec <path> | exec-any <basename> | exists <path>
#
# Put the Buildroot symbol in `why`, so a failure is one line away from its fix.

# --- Health and remote access (US-SRV-3, US-SRV-5) ------------------------------
exec-any  smartctl                                 # SMART disk health — BR2_PACKAGE_SMARTMONTOOLS
exec-any  nvme                                     # NVMe health log — BR2_PACKAGE_NVME
exec-any  wg                                       # WireGuard keys and status — BR2_PACKAGE_WIREGUARD_TOOLS
exec-any  ethtool                                  # NIC link, offloads, Wake-on-LAN — BR2_PACKAGE_ETHTOOL
exec-any  nvtop                                    # GPU monitor (NVIDIA, AMD, Intel) — BR2_PACKAGE_NVTOP

# --- GPU firmware for xe (US-SRV-9 transcoding) ---------------------------------
exists    usr/lib/firmware/xe/bmg_guc_70.bin       # Battlemage GuC — BR2_PACKAGE_LINUX_FIRMWARE_XE

# --- What manylinux wheels assume the host has (US-SRV-10) ----------------------
exists    usr/lib/libGL.so.1                       # libglvnd GLX dispatch — BR2_PENGYOS_MANYLINUX_LIBS
exists    usr/lib/libEGL.so.1                      # libglvnd EGL dispatch — BR2_PENGYOS_MANYLINUX_LIBS
exists    usr/lib/libX11.so.6                      # BR2_PENGYOS_MANYLINUX_LIBS
exists    usr/lib/libXext.so.6                     # BR2_PENGYOS_MANYLINUX_LIBS
exists    usr/lib/libXrender.so.1                  # BR2_PENGYOS_MANYLINUX_LIBS
exists    usr/lib/libSM.so.6                       # BR2_PENGYOS_MANYLINUX_LIBS
exists    usr/lib/libICE.so.6                      # BR2_PENGYOS_MANYLINUX_LIBS

# --- SMB, both directions, without Samba (US-SRV-13, US-SRV-14) -----------------
exec-any  mount.cifs                               # fstab `credentials=` (the kernel ignores it) — BR2_PACKAGE_CIFS_UTILS
exec-any  ksmbd.mountd                             # the SMB3 server's user/share daemon — BR2_PACKAGE_KSMBD_TOOLS
exec-any  ksmbd.control                            # reload/shutdown, ksmbd.service's ExecReload — BR2_PACKAGE_KSMBD_TOOLS
exists    usr/lib/systemd/system/ksmbd.service     # pengyos-shares enables it — BR2_PACKAGE_KSMBD_TOOLS (systemd unit dir)
exec      usr/bin/pengyos-shares                   # board/server overlay
exec      usr/lib/systemd/systemd-logind           # power button; sessions — BR2_PACKAGE_SYSTEMD_LOGIND (core layer)
exec      usr/lib/polkit-1/polkitd                 # logind asks it for non-root callers — BR2_PACKAGE_POLKIT (server layer)
contains  usr/share/polkit-1/actions/org.freedesktop.login1.policy  set-self-linger   # systemd built -Dpolkit=enabled (external.mk)
exists    etc/polkit-1/rules.d/60-pengyos-linger.rules   # users set their own linger — server overlay
exec      usr/bin/pengyos-vm                       # board/server overlay
exists    usr/lib/systemd/user/pengyos-vm@.service   # board/server overlay
exec      usr/libexec/pengyos/pengyos-vm-shutdown-notice   # says on the console which VMs a shutdown waits for
exec      usr/bin/pengy-vmm                        # board/server overlay
exists    usr/lib/systemd/user/pengy-vmm.service   # board/server overlay
exec      usr/bin/qemu-system-x86_64               # pengyos-vm runs it — BR2_PACKAGE_QEMU_SYSTEM
contains  usr/bin/qemu-system-x86_64  vnc-listen   # -vnc (pengyos-vm vnc) — BR2_PACKAGE_LIBVNCSERVER turns on QEMU's VNC
exec      usr/bin/qemu-img                         # pengyos-vm create — BR2_PACKAGE_QEMU_TOOLS
exists    usr/lib/systemd/user/podman.socket       # rootless API socket — external.mk PENGYOS_PODMAN_INSTALL_UNITS
contains  usr/lib/systemd/user/podman.service  /usr/bin/podman   # @@PODMAN@@ substituted
exists    usr/lib/systemd/user/podman-restart.service   # rootless --restart=always after a reboot
contains  usr/lib/systemd/system/podman-restart.service  /usr/bin/podman   # the same for root containers
exists    usr/share/qemu/edk2-x86_64-code.fd       # UEFI guests — BR2_PACKAGE_QEMU_BLOBS
exists    usr/share/qemu/edk2-i386-vars.fd         # UEFI variables template — BR2_PACKAGE_QEMU_BLOBS
exec      usr/libexec/qemu-bridge-helper           # --net bridge (setuid: board/server/permissions.txt)

# --- Containers at boot (US-SRV-9) ---
exec      usr/libexec/podman/quadlet                       # .container files -> services — external.mk PENGYOS_PODMAN_INSTALL_QUADLET
exec      usr/lib/systemd/user-generators/podman-user-generator      # rootless (~/.config/containers/systemd) — same hook
exec      usr/lib/systemd/system-generators/podman-system-generator  # rootful (/etc/containers/systemd) — same hook
exists    etc/containers/systemd/README                              # the system Quadlet lane, explained where it is used — board/server/rootfs-overlay
exec      usr/libexec/podman/rootlessport                  # -p on a user-defined network — external.mk PENGYOS_PODMAN_INSTALL_ROOTLESSPORT

# --- Observability (Pengy's Server review, R3) -----------------------------------
exists    usr/bin/pengyos-services                 # board/core overlay; what runs, autostarts, listens, is reachable

# --- AMD GPU compute in containers (Strix Halo, 2026-11-04) ----------------------
# ROCm lives in the container; the host gives it /dev/kfd (kernel: server
# linux.fragment HSA_AMD), crun and no locked-memory limit.
exec-any  crun                                     # rootless --group-add keep-groups — BR2_PACKAGE_CRUN
exists    usr/lib/systemd/system.conf.d/60-pengyos-memlock.conf  # --ulimit memlock=-1 in rootless containers
exists    usr/lib/systemd/user.conf.d/60-pengyos-memlock.conf    # ...and in a user's Quadlets
exists    usr/bin/pengyos-gpu-memory               # board/server overlay; the APU's GTT limit (unified memory)

# --- Python apps on the host: uv (Pengy's field review R4, 2026-11-05) -----------
# One venv per app against the system python3 (US-SRV-10). Wheels that need
# compiling are built in the dev guest; pengyos.txt has the recipe.
exec      usr/bin/uv                               # BR2_PACKAGE_UV (buildroot-external/package/uv)
exec      usr/bin/uvx                              # uv's tool runner, from the same crate
